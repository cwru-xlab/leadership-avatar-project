```ts
export const MAX_DEMONSTRATIONS_PER_LEG = 1;
export const MAX_DEMONSTRATION_LENGTH = 400;   // PROVISIONAL UNTIL CALIBRATED
export const MAX_COACH_SPEECH_LENGTH = 700;    // PROVISIONAL UNTIL CALIBRATED
export const DEMONSTRATION_REJECT_REASONS = ["DEMONSTRATION_BEFORE_ATTEMPT", "DEMONSTRATION_LIMIT", "TOO_LONG", "ECHOES_STUDENT_LINE"] as const;
export const COACH_SPEECH_REJECT_REASONS = ["CONTAINS_SCRIPT_CUE", "CONTAINS_SCRIPTED_QUOTE", "VOICES_COUNTERPART", "TOO_LONG"] as const;

export type TipGate = { kind: "DEMONSTRATION"; attemptsInLeg: number; demonstrationsInLeg: number };

export function validateDemonstration(text: string, ctx: { studentLine: string; attemptsInLeg: number; demonstrationsInLeg: number }):
  { ok: true } | { ok: false; reason: (typeof DEMONSTRATION_REJECT_REASONS)[number]; detail: string };

// UNCHANGED signature for existing callers; optional gate added.
export function assertTipSafe(tip: string, context: { studentLine: string; gate?: TipGate }): void;

export function validateCoachSpeech(speech: string, ctx: { studentLine: string; counterpartLine: string | null; studentAttempts: readonly string[] }):
  { ok: true } | { ok: false; reason: (typeof COACH_SPEECH_REJECT_REASONS)[number]; detail: string };

export function isParrotOf(attempt: string, demonstration: string): boolean;
```

Rules:
- `assertTipSafe` without `gate` → exactly today's `validateTip` behaviour.
- With `gate.kind === "DEMONSTRATION"` → `validateDemonstration`; throws with the
  reason. Order: `attemptsInLeg < 1` → DEMONSTRATION_BEFORE_ATTEMPT (checked FIRST,
  before any content rule — the whole point); `demonstrationsInLeg >=
  MAX_DEMONSTRATIONS_PER_LEG` → DEMONSTRATION_LIMIT; length > 400 → TOO_LONG;
  a 10+ word contiguous span shared with `studentLine` → ECHOES_STUDENT_LINE
  (higher than the tip rule's 6 because a better version legitimately reuses some
  of the student's words — comment that). Quoted speech, script cues and
  first-person sentences are WAIVED for a gated demonstration — a demonstration
  is, by design, words to say.
- `validateCoachSpeech`: SCRIPT_CUE_PHRASES (reuse the existing exported list)
  → CONTAINS_SCRIPT_CUE; a quoted span of 4+ words that is NOT a substring of the
  student's original line or of any of `studentAttempts` → CONTAINS_SCRIPTED_QUOTE
  (quoting the student's own words back is coaching; quoting new words is
  scripting); a 6+ word contiguous span shared with `counterpartLine` →
  VOICES_COUNTERPART (22-CONTEXT.md: the other party is "never voiced by the
  coach"); length > 700 → TOO_LONG. The first-person rule does NOT apply (the
  coach speaks as "I").
- `isParrotOf(attempt, demonstration)`: true when they share a contiguous 8+ word
  span, OR when ≥ 70% of the demonstration's word tokens (lowercased, punctuation
  stripped) appear in order-insensitive token overlap with the attempt AND the
  attempt is no more than 1.3× the demonstration's length. PROVISIONAL UNTIL
  CALIBRATED; comment the false-positive tradeoff in the same voice as the module
  header.
